cmake_minimum_required(VERSION 3.16)
project(rkllm_stream LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RKLLM_INCLUDE_DIR "" CACHE PATH "Official matching rkllm.h include directory")
set(RKLLM_LIB "" CACHE FILEPATH "Matching ARM64 librkllmrt.so")
if(NOT EXISTS "${RKLLM_INCLUDE_DIR}/rkllm.h")
  message(FATAL_ERROR "Provide -DRKLLM_INCLUDE_DIR=/sdk/include")
endif()
if(NOT EXISTS "${RKLLM_LIB}")
  message(FATAL_ERROR "Provide -DRKLLM_LIB=/sdk/aarch64/librkllmrt.so")
endif()
add_executable(rkllm_stream rkllm_stream.cpp)
target_include_directories(rkllm_stream PRIVATE "${RKLLM_INCLUDE_DIR}")
target_link_libraries(rkllm_stream PRIVATE "${RKLLM_LIB}")
set_target_properties(rkllm_stream PROPERTIES BUILD_RPATH "$ORIGIN")
